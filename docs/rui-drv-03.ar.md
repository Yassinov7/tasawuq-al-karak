```sql
select to_regprocedure('public.read_driver_money(text,text,timestamp with time zone,uuid)') is not null as driver_read_ready,
       has_function_privilege('authenticated', 'public.read_driver_money(text,text,timestamp with time zone,uuid)', 'execute') as authenticated_execute,
       not has_function_privilege('anon', 'public.read_driver_money(text,text,timestamp with time zone,uuid)', 'execute') as anon_denied;
```
